9async fn fetch(file_name: &str) -> Option<String> {
10    use wasm_bindgen::prelude::*;
11    use wasm_bindgen_futures::JsFuture;
12    use web_sys::{Request, RequestInit, RequestMode, Response};
13    let window = web_sys::window().unwrap();
14    let location = window.location().origin().ok()?;
15    let url = format!("{location}/{file_name}");
16    let opts = RequestInit::new();
17    opts.set_mode(RequestMode::Cors);
18
19    let request = Request::new_with_str_and_init(&url, &opts).ok()?;
20
21    let resp_value = JsFuture::from(window.fetch_with_request(&request))
22        .await
23        .ok()?;
24
25    // `resp_value` is a `Response` object.
26    assert!(resp_value.is_instance_of::<Response>());
27    let resp: Response = resp_value.dyn_into().unwrap();
28
29    // Convert this other `Promise` into a rust `Future`.
30    let json = JsFuture::from(resp.text().ok()?).await.ok()?;
31    json.as_string()
32}
33async fn load_string(file_name: &str) -> Option<String> {
34    cfg_if! {
35        if #[cfg(target_arch = "wasm32")] {
36            fetch(file_name).await
37
38        } else {
39            let path = std::path::Path::new(file_name);
40            let text = std::fs::read_to_string(path).ok()?;
41            Some(text)
42        }
43    }
44}

46/// Helper to load a mesh from an obj file
47pub async fn load_mesh(file_name: &str) -> Option<(Vec<[f32; 3]>, SurfaceIndices)> {
48    let obj_text = load_string(file_name).await?;
49    let obj_cursor = Cursor::new(obj_text);
50    let mut obj_reader = BufReader::new(obj_cursor);
51    Some(crate::obj_load::load_obj_buf(&mut obj_reader))
52}
